function light(flameY) {
  const span = rowH * reachRows;
  links.forEach((link, i) => {
    const d = Math.abs(rows[i]
      - flameY) / span;
    const b = Math.round(clamp(
      1 - d ** 1.5, 0, 1) * 40) / 40;
  });
}
